cmake_minimum_required(VERSION 3.31)
project(GhostUISony LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The GhostUI Sony bridge currently targets macOS")
endif()
if(NOT SONY_SOURCE_DIR)
  message(FATAL_ERROR "Run tools/setup_sony.py to supply the pinned Sony source")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(MDR_BUILD_CLIENT OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_CODEGEN OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_WITH_IPO OFF CACHE BOOL "" FORCE)
set(MDR_BT_BLE_SUPPORT OFF CACHE BOOL "" FORCE)
set(MDR_ENABLE_LOG OFF CACHE BOOL "" FORCE)
add_subdirectory("${SONY_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)
add_executable(ghostui-sony SonyBridge.mm)
target_compile_options(ghostui-sony PRIVATE -fobjc-arc)
target_link_libraries(ghostui-sony PRIVATE mdr mdr-bt "-framework Foundation")
target_link_options(ghostui-sony PRIVATE "-Wl,-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
